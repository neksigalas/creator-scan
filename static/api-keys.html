import requests

API_KEY = "cs_your_key_here"
BASE    = "http://localhost:8000"

# Βρες Gaming creators στο YouTube με 5K-50K followers
resp = requests.get(
    f"{BASE}/v1/creators",
    headers={"X-API-Key": API_KEY},
    params={
        "platform":      "youtube",
        "niche":         "Gaming",
        "min_followers":  5000,
        "max_followers":  50000,
        "has_email":      "true",
        "limit":          20,
    }
)
data = resp.json()
print(f"Βρέθηκαν {data['total']} creators")
for c in data["data"]:
    print(f"{c['display_name']} — {c['followers']:,} followers — {c['email']}")

const API_KEY = "cs_your_key_here";
const BASE    = "http://localhost:8000";

// Βρες Tech creators με email
const resp = await fetch(
  `${BASE}/v1/creators?niche=Tech&has_email=true&limit=50`,
  { headers: { "X-API-Key": API_KEY } }
);
const { data, total } = await resp.json();
console.log(`${total} creators found`, data);
